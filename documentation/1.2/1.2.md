```dart
import 'package:flutter/material.dart';

class MyAdaptiveApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    final size = MediaQuery.of(context).size;
    final config = AdaptiveConfig.fromDimensions(
      width: size.width,
      height: size.height,
    );

    // Debug info
    print(config.toDebugString());

    return Scaffold(
      body: Builder(
        builder: (context) {
          // Layout baseado em window size class
          if (config.isCompact) {
            return _buildCompactLayout(config);
          } else if (config.isMedium) {
            return _buildMediumLayout(config);
          } else {
            return _buildExpandedLayout(config);
          }
        },
      ),
    );
  }

  Widget _buildExpandedLayout(AdaptiveConfig config) {
    return Row(
      children: [
        // Navigation Rail or Drawer
        if (AdaptiveUtils.shouldUsePermanentDrawer(config.sizeClass))
          SizedBox(
            width: AdaptiveUtils.getNavigationDrawerWidth(config.sizeClass),
            child: NavigationDrawer(),
          )
        else
          NavigationRail(extended: false),

        // Main content with max width
        Expanded(
          child: Center(
            child: ConstrainedBox(
              constraints: BoxConstraints(
                maxWidth: config.maxContentWidth,
              ),
              child: MainContent(),
            ),
          ),
        ),

        // Supporting pane if space allows
        if (config.supportsThreePane)
          SizedBox(
            width: AdaptiveUtils.getSupportingPaneWidth(
              totalWidth: config.width,
              sizeClass: config.sizeClass,
            ),
            child: SupportingPane(),
          ),
      ],
    );
  }
}
```
